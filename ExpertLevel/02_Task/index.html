Sass Features Used

// Variables
$primary-color: #3498db;
$font-size-base: 16px;

// Nesting
.navbar {
  &__brand { ... }
  &__menu { ... }
  &__link {
    &:hover { ... }
  }
}

// Mixins
@mixin flex-center {
  display: flex;
  justify-content: center;
  align-items: center;
}

// Functions
background: darken($primary-color, 10%);
